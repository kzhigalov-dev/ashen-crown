`js/Level.js`:

```js
import { TILE } from './data/constants.js';

export const EMPTY = 0;
export const SOLID = 1;
export const ONEWAY = 2;
export const HAZARD = 3;

const TILE_CHARS = { '.': EMPTY, '#': SOLID, '=': ONEWAY, '^': HAZARD, '~': HAZARD };

// Склеивает сегменты уровня слева направо в одну карту.
export function joinSegments(segments) {
    const rows = segments[0].length;
    segments.forEach((segment, index) => {
        if (segment.length !== rows) {
            throw new Error(`Segment ${index}: ${segment.length} rows, expected ${rows}`);
        }
        segment.forEach((line, row) => {
            if (line.length !== segment[0].length) {
                throw new Error(
                    `Segment ${index}, row ${row}: length ${line.length}, expected ${segment[0].length}`
                );
            }
        });
    });
    return Array.from({ length: rows }, (_, row) => segments.map((segment) => segment[row]).join(''));
}

export class Level {
    constructor(def) {
        this.id = def.id;
        this.name = def.name;
        this.chars = joinSegments(def.segments);
        this.rows = this.chars.length;
        this.cols = this.chars[0].length;
        this.tiles = [];
        this.spawn = null;
        this.exit = null;
        this.checkpoints = [];
        this.coins = [];

        this.chars.forEach((line, row) => {
            const tileRow = [];
            for (let col = 0; col < line.length; col++) {
                const char = line[col];
                const point = { x: col * TILE, y: row * TILE, w: TILE, h: TILE };
                if (char in TILE_CHARS) {
                    tileRow.push(TILE_CHARS[char]);
                    continue;
                }
                tileRow.push(EMPTY);
                if (char === 'P') this.spawn = point;
                else if (char === 'E') this.exit = point;
                else if (char === 'C') this.checkpoints.push(point);
                else if (char === 'o') this.coins.push(point);
                else throw new Error(`Level "${def.id}": unknown char "${char}" at ${col},${row}`);
            }
            this.tiles.push(tileRow);
        });

        if (!this.spawn) throw new Error(`Level "${def.id}": no spawn point (P)`);
        if (!this.exit) throw new Error(`Level "${def.id}": no exit point (E)`);
    }

    get width() {
        return this.cols * TILE;
    }

    get height() {
        return this.rows * TILE;
    }

    // За боковыми краями — стена, выше и ниже уровня — пусто (в яму можно упасть).
    tileAt(col, row) {
        if (col < 0 || col >= this.cols) return SOLID;
        if (row < 0 || row >= this.rows) return EMPTY;
        return this.tiles[row][col];
    }

    charAt(col, row) {
        if (col < 0 || col >= this.cols || row < 0 || row >= this.rows) return '.';
        return this.chars[row][col];
    }
}
```

`js/Collision.js`:

```js
import { TILE } from './data/constants.js';
import { SOLID, ONEWAY } from './Level.js';

export function overlaps(a, b) {
    return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
}

// Номера крайних тайлов, которые занимает отрезок [start, start + size).
// Правая и нижняя границы не включаются: тело шириной 16 в x=0 занимает один тайл.
function tileRange(start, size) {
    return [Math.floor(start / TILE), Math.ceil((start + size) / TILE) - 1];
}

// Двигает тело и выталкивает его из сплошных тайлов. Оси обрабатываются
// раздельно: сначала X, потом Y — так тело не цепляется за углы.
// Скорости ограничены так, что за шаг тело проходит меньше одного тайла.
export function moveAndCollide(body, level, dt) {
    const result = { onGround: false, hitWall: false, hitCeiling: false };

    body.x += body.vx * dt;
    if (body.vx !== 0) {
        const [top, bottom] = tileRange(body.y, body.h);
        const col = body.vx > 0 ? tileRange(body.x, body.w)[1] : Math.floor(body.x / TILE);
        for (let row = top; row <= bottom; row++) {
            if (level.tileAt(col, row) !== SOLID) continue;
            body.x = body.vx > 0 ? col * TILE - body.w : (col + 1) * TILE;
            body.vx = 0;
            result.hitWall = true;
            break;
        }
    }

    const prevBottom = body.y + body.h;
    body.y += body.vy * dt;
    const [left, right] = tileRange(body.x, body.w);
    if (body.vy > 0) {
        const row = tileRange(body.y, body.h)[1];
        const tileTop = row * TILE;
        for (let col = left; col <= right; col++) {
            const tile = level.tileAt(col, row);
            // Платформа держит, только если тело было выше её верха до шага.
            const lands = tile === SOLID || (tile === ONEWAY && prevBottom <= tileTop);
            if (!lands) continue;
            body.y = tileTop - body.h;
            body.vy = 0;
            result.onGround = true;
            break;
        }
    } else if (body.vy < 0) {
        const row = Math.floor(body.y / TILE);
        for (let col = left; col <= right; col++) {
            if (level.tileAt(col, row) !== SOLID) continue;
            body.y = (row + 1) * TILE;
            body.vy = 0;
            result.hitCeiling = true;
            break;
        }
    }

    return result;
}

export function touchesTile(body, level, type) {
    const [left, right] = tileRange(body.x, body.w);
    const [top, bottom] = tileRange(body.y, body.h);
    for (let row = top; row <= bottom; row++) {
        for (let col = left; col <= right; col++) {
            if (level.tileAt(col, row) === type) return true;
        }
    }
    return false;
}
```

```js
import { Entity } from './Entity.js';
import { moveAndCollide } from '../Collision.js';
import {
    TILE, GRAVITY, RUN_SPEED, JUMP_SPEED, MAX_FALL,
    COYOTE_TIME, JUMP_BUFFER, JUMP_CUT, INVULN_TIME,
} from '../data/constants.js';

export class Player extends Entity {
    constructor() {
        super(0, 0, 10, 22);
        this.maxHp = 100;
        this.hp = 100;
        this.coyote = 0;      // сколько ещё можно прыгнуть после схода с края
        this.jumpBuffer = 0;  // сколько ещё помнится нажатие прыжка
        this.invuln = 0;      // остаток неуязвимости после урона
        this.animTime = 0;
        this.state = 'idle';
    }

    update(dt, input, level) {
        const dir = (input.isDown('right') ? 1 : 0) - (input.isDown('left') ? 1 : 0);
        this.vx = dir * RUN_SPEED;
        if (dir !== 0) this.facing = dir;

        // onGround здесь — результат прошлого шага.
        this.coyote = this.onGround ? COYOTE_TIME : Math.max(0, this.coyote - dt);
        this.jumpBuffer = input.wasPressed('jump') ? JUMP_BUFFER : Math.max(0, this.jumpBuffer - dt);

        if (this.jumpBuffer > 0 && this.coyote > 0) {
            this.vy = -JUMP_SPEED;
            this.jumpBuffer = 0;
            this.coyote = 0;
        }

        // Клавиша отпущена на взлёте — срезаем скорость, прыжок получается ниже.
        const cutSpeed = -JUMP_SPEED * JUMP_CUT;
        if (!input.isDown('jump') && this.vy < cutSpeed) this.vy = cutSpeed;

        this.vy = Math.min(this.vy + GRAVITY * dt, MAX_FALL);
        this.onGround = moveAndCollide(this, level, dt).onGround;

        this.invuln = Math.max(0, this.invuln - dt);
        this.animTime += dt;
        if (!this.onGround) this.state = 'jump';
        else this.state = dir !== 0 ? 'run' : 'idle';
    }

    // Урон от врагов и ловушек. Возвращает false, если урон не прошёл.
    hurt(amount) {
        if (this.dead || this.invuln > 0) return false;
        this.takeDamage(amount);
        this.invuln = INVULN_TIME;
        return true;
    }

    // Ставит героя ногами на нижний край тайла, по центру.
    placeAt(point) {
        this.x = point.x + (TILE - this.w) / 2;
        this.y = point.y + TILE - this.h;
        this.vx = 0;
        this.vy = 0;
    }
}
```

```js
// Подмена window для тестов: хранит обработчики и умеет рассылать события.
export function fakeTarget() {
    const listeners = new Map();
    return {
        addEventListener(type, fn) {
            if (!listeners.has(type)) listeners.set(type, new Set());
            listeners.get(type).add(fn);
        },
        removeEventListener(type, fn) {
            listeners.get(type)?.delete(fn);
        },
        dispatch(type, event = {}) {
            const full = { repeat: false, preventDefault() {}, ...event };
            for (const fn of listeners.get(type) ?? []) fn(full);
        },
        count() {
            let total = 0;
            for (const set of listeners.values()) total += set.size;
            return total;
        },
    };
}
```

```js
// Клавиши задаются через event.code: это физическая клавиша,
// поэтому управление не зависит от раскладки (русская или английская).
export const KEY_MAP = {
    KeyA: 'left',
    ArrowLeft: 'left',
    KeyD: 'right',
    ArrowRight: 'right',
    Space: 'jump',
    KeyW: 'jump',
    ArrowUp: 'jump',
    KeyJ: 'attack',
    KeyK: 'ability',
    KeyP: 'pause',
    Escape: 'pause',
};

export class InputHandler {
    constructor(target) {
        this.target = target;
        this.codes = new Set();    // удерживаемые клавиши
        this.pressed = new Set();  // действия, нажатые с прошлого шага
        this.capture = false;      // true — гасим прокрутку страницы и т. п.

        this.onKeyDown = (event) => {
            const action = KEY_MAP[event.code];
            if (!action) return;
            if (this.capture) event.preventDefault();
            if (event.repeat) return;
            this.codes.add(event.code);
            this.pressed.add(action);
        };
        this.onKeyUp = (event) => {
            if (!KEY_MAP[event.code]) return;
            if (this.capture) event.preventDefault();
            this.codes.delete(event.code);
        };
        this.onBlur = () => this.reset();
    }

    attach() {
        this.target.addEventListener('keydown', this.onKeyDown);
        this.target.addEventListener('keyup', this.onKeyUp);
        this.target.addEventListener('blur', this.onBlur);
    }

    detach() {
        this.target.removeEventListener('keydown', this.onKeyDown);
        this.target.removeEventListener('keyup', this.onKeyUp);
        this.target.removeEventListener('blur', this.onBlur);
        this.reset();
    }

    reset() {
        this.codes.clear();
        this.pressed.clear();
    }

    isDown(action) {
        for (const code of this.codes) {
            if (KEY_MAP[code] === action) return true;
        }
        return false;
    }

    wasPressed(action) {
        return this.pressed.has(action);
    }

    consume(action) {
        return this.pressed.delete(action);
    }

    endStep() {
        this.pressed.clear();
    }
}
```

```js
import { Level, HAZARD } from './Level.js';
import { Player } from './entities/Player.js';
import { overlaps, touchesTile } from './Collision.js';
import {
    STEP, MAX_FRAME, VIEW_W, VIEW_H,
    HAZARD_DAMAGE, INVULN_TIME, COIN_VALUE, CAMERA_LERP,
} from './data/constants.js';

export class Game {
    constructor({ levelDef, input, renderer = null, onStateChange = () => {} }) {
        this.levelDef = levelDef;
        this.input = input;
        this.renderer = renderer;
        this.onStateChange = onStateChange;

        this.state = 'menu';
        this.level = null;
        this.player = null;
        this.coins = [];
        this.checkpoint = null;
        this.gold = 0;
        this.time = 0;
        this.camera = { x: 0, y: 0 };

        this.rafId = null;
        this.last = null;       // время прошлого кадра, мс
        this.accumulator = 0;   // накопленное, но ещё не отыгранное время, с

        this.frame = (now) => {
            this.tick(now);
            this.rafId = requestAnimationFrame(this.frame);
        };
    }

    setState(state) {
        this.state = state;
        this.input.capture = state === 'playing';
        this.onStateChange(state, this);
    }

    startLevel() {
        this.level = new Level(this.levelDef);
        this.player = new Player();
        this.player.placeAt(this.level.spawn);
        this.checkpoint = this.level.spawn;
        this.coins = this.level.coins.map((coin) => ({ ...coin, taken: false }));
        this.gold = 0;
        this.time = 0;
        this.accumulator = 0;
        this.last = null;
        this.camera = { x: this.cameraTarget(), y: Math.max(0, this.level.height - VIEW_H) };
        this.setState('playing');
    }

    // Запуск цикла и подключение клавиатуры.
    start() {
        this.input.attach();
        this.last = null;
        this.rafId = requestAnimationFrame(this.frame);
    }

    // Остановка цикла и снятие обработчиков.
    stop() {
        if (this.rafId !== null) cancelAnimationFrame(this.rafId);
        this.rafId = null;
        this.input.detach();
    }

    restart() {
        this.stop();
        this.startLevel();
        this.start();
    }

    toMenu() {
        this.stop();
        this.level = null;
        this.setState('menu');
    }

    pause() {
        if (this.state === 'playing') this.setState('paused');
    }

    resume() {
        if (this.state !== 'paused') return;
        this.last = null;
        this.setState('playing');
    }

    togglePause() {
        if (this.state === 'playing') this.pause();
        else this.resume();
    }

    // Один кадр. Реальное время копится и расходуется шагами по STEP,
    // поэтому физика одинакова на любом мониторе.
    tick(now) {
        // Пауза обрабатывается до подсчёта времени: resume() сбрасывает last,
        // и время, проведённое на паузе, не попадает в логику.
        if (this.input.consume('pause')) this.togglePause();

        const delta = this.last === null ? 0 : Math.min((now - this.last) / 1000, MAX_FRAME);
        this.last = now;

        if (this.state === 'playing') {
            this.accumulator += delta;
            while (this.accumulator >= STEP && this.state === 'playing') {
                this.update(STEP);
                this.input.endStep();
                this.accumulator -= STEP;
            }
        } else {
            this.accumulator = 0;
            this.input.endStep();
        }

        if (this.renderer && this.level) this.renderer.draw(this);
    }

    update(dt) {
        const { player, level } = this;
        player.update(dt, this.input, level);
        this.time += dt;

        for (const coin of this.coins) {
            if (coin.taken || !overlaps(player, coin)) continue;
            coin.taken = true;
            this.gold += COIN_VALUE;
        }

        for (const checkpoint of level.checkpoints) {
            if (overlaps(player, checkpoint)) this.checkpoint = checkpoint;
        }

        const fellOut = player.y > level.height;
        if (fellOut || touchesTile(player, level, HAZARD)) {
            player.takeDamage(HAZARD_DAMAGE);
            player.invuln = INVULN_TIME;
            if (!player.dead) player.placeAt(this.checkpoint);
        }

        this.camera.x += (this.cameraTarget() - this.camera.x) * Math.min(1, CAMERA_LERP * dt);

        if (player.dead) this.setState('dead');
        else if (overlaps(player, level.exit)) this.setState('complete');
    }

    cameraTarget() {
        const center = this.player.x + this.player.w / 2 - VIEW_W / 2;
        return Math.max(0, Math.min(center, this.level.width - VIEW_W));
    }
}
```

```js
import { TILE, VIEW_W, VIEW_H } from './data/constants.js';
import { SOLID } from './Level.js';
import { PALETTE, SPRITES } from './data/sprites.js';
import { drawHud } from './ui/Hud.js';

const SKY = '#141b2b';
const FAR_TREES = '#1a2c33';
const NEAR_TREES = '#21403a';
const TREE_HEIGHTS = [38, 62, 46, 80, 54, 70, 42, 66];
const TREE_GAP = 44;

// Рисует спрайт из строк палитры во внеэкранный canvas (один раз при запуске).
function buildSprite(rows, flip) {
    const canvas = document.createElement('canvas');
    canvas.width = rows[0].length;
    canvas.height = rows.length;
    const ctx = canvas.getContext('2d');
    rows.forEach((line, y) => {
        for (let x = 0; x < line.length; x++) {
            const color = PALETTE[line[x]];
            if (!color) continue;
            ctx.fillStyle = color;
            ctx.fillRect(flip ? line.length - 1 - x : x, y, 1, 1);
        }
    });
    return canvas;
}

export class Renderer {
    constructor(canvas) {
        this.canvas = canvas;
        canvas.width = VIEW_W;
        canvas.height = VIEW_H;
        this.ctx = canvas.getContext('2d');
        this.ctx.imageSmoothingEnabled = false;

        this.sprites = {};
        this.flipped = {};
        for (const [name, rows] of Object.entries(SPRITES)) {
            this.sprites[name] = buildSprite(rows, false);
            this.flipped[name] = buildSprite(rows, true);
        }
    }

    // Увеличивает поле в целое число раз, чтобы пиксели оставались чёткими.
    resize() {
        const scale = Math.max(
            1,
            Math.floor(Math.min(window.innerWidth / VIEW_W, window.innerHeight / VIEW_H))
        );
        const stage = this.canvas.parentElement;
        stage.style.width = `${VIEW_W * scale}px`;
        stage.style.height = `${VIEW_H * scale}px`;
        stage.style.setProperty('--scale', scale);
    }

    draw(game) {
        const { ctx } = this;
        const camX = Math.round(game.camera.x);
        const camY = Math.round(game.camera.y);

        ctx.fillStyle = SKY;
        ctx.fillRect(0, 0, VIEW_W, VIEW_H);
        // Дальний слой движется медленнее ближнего — это и есть параллакс.
        this.drawTrees(camX * 0.2, FAR_TREES, 0.6);
        this.drawTrees(camX * 0.5, NEAR_TREES, 1);

        ctx.save();
        ctx.translate(-camX, -camY);
        this.drawTiles(game.level, camX);
        this.drawObjects(game);
        this.drawPlayer(game.player);
        ctx.restore();

        drawHud(ctx, game, this.sprites);
    }

    drawTrees(offset, color, k) {
        const { ctx } = this;
        const base = VIEW_H - 30;
        ctx.fillStyle = color;
        ctx.fillRect(0, base, VIEW_W, 30);
        const first = Math.floor(offset / TREE_GAP);
        for (let i = first - 1; i <= first + VIEW_W / TREE_GAP + 1; i++) {
            const index = ((i % TREE_HEIGHTS.length) + TREE_HEIGHTS.length) % TREE_HEIGHTS.length;
            const height = Math.round(TREE_HEIGHTS[index] * k);
            const x = Math.round(i * TREE_GAP - offset);
            const trunk = Math.round(height * 0.4);
            const tier = Math.round(height * 0.15);
            ctx.fillRect(x + 9, base - trunk, 4, trunk);
            for (let t = 0; t < 4; t++) {
                const width = 22 - t * 5;
                ctx.fillRect(x + 11 - Math.round(width / 2), base - trunk - (t + 1) * tier, width, tier + 1);
            }
        }
    }

    drawTiles(level, camX) {
        const { ctx, sprites } = this;
        const first = Math.max(0, Math.floor(camX / TILE));
        const last = Math.min(level.cols - 1, Math.floor((camX + VIEW_W) / TILE));
        for (let row = 0; row < level.rows; row++) {
            for (let col = first; col <= last; col++) {
                const char = level.charAt(col, row);
                const x = col * TILE;
                const y = row * TILE;
                if (char === '#') {
                    const covered = level.tileAt(col, row - 1) === SOLID;
                    ctx.drawImage(covered ? sprites.ground : sprites.groundTop, x, y);
                } else if (char === '=') {
                    ctx.drawImage(sprites.platform, x, y);
                } else if (char === '^') {
                    ctx.drawImage(sprites.spikes, x, y + TILE - sprites.spikes.height);
                }
            }
        }
    }

    drawObjects(game) {
        const { ctx, sprites } = this;
        for (const checkpoint of game.level.checkpoints) {
            const lit = checkpoint === game.checkpoint;
            const sprite = lit ? sprites.campfire : sprites.campfireOff;
            ctx.drawImage(sprite, checkpoint.x + 3, checkpoint.y + TILE - sprite.height);
        }
        ctx.drawImage(sprites.portal, game.level.exit.x + 2, game.level.exit.y);
        for (const coin of game.coins) {
            if (coin.taken) continue;
            const bob = Math.round(Math.sin(game.time * 4 + coin.x) * 1.5);
            ctx.drawImage(sprites.coin, coin.x + 4, coin.y + 4 + bob);
        }
    }

    drawPlayer(player) {
        // Мигание во время неуязвимости.
        if (player.invuln > 0 && Math.floor(player.invuln * 20) % 2 === 0) return;

        let name = 'heroIdle';
        if (player.state === 'jump') name = 'heroJump';
        else if (player.state === 'run') name = Math.floor(player.animTime * 10) % 2 ? 'heroRun1' : 'heroRun2';

        const right = player.facing === 1;
        const sprite = right ? this.sprites[name] : this.flipped[name];
        // Спрайт шире хитбокса: лишние пиксели (рука) торчат в сторону взгляда.
        const x = right ? player.x : player.x - (sprite.width - player.w);
        this.ctx.drawImage(sprite, Math.round(x), Math.round(player.y));
    }
}
```

`js/ui/Screens.js`:

```js
// HTML-экраны поверх игрового поля: меню, пауза, победа, поражение.
export class Screens {
    constructor(root) {
        this.root = root;
        this.screens = {
            menu: root.getElementById('screen-menu'),
            paused: root.getElementById('screen-paused'),
            complete: root.getElementById('screen-complete'),
            dead: root.getElementById('screen-dead'),
        };
        this.pauseButton = root.getElementById('btn-pause');
    }

    // handlers: { start, resume, restart, pause, menu } — по значению data-action.
    bind(handlers) {
        for (const button of this.root.querySelectorAll('[data-action]')) {
            button.addEventListener('click', () => handlers[button.dataset.action]());
        }
    }

    onState(state, game) {
        for (const [name, element] of Object.entries(this.screens)) {
            element.hidden = name !== state;
        }
        this.pauseButton.hidden = state !== 'playing';
        for (const element of this.root.querySelectorAll('[data-gold]')) {
            element.textContent = game.gold;
        }
        // Фокус на первую кнопку экрана — им можно управлять с клавиатуры.
        const active = this.screens[state];
        if (active) active.querySelector('button').focus();
    }
}
```

`js/main.js`:

```js
import { Game } from './Game.js';
import { InputHandler } from './InputHandler.js';
import { Renderer } from './Renderer.js';
import { Screens } from './ui/Screens.js';
import { LEVELS } from './data/levels.js';

const renderer = new Renderer(document.getElementById('game'));
const input = new InputHandler(window);
const screens = new Screens(document);

const game = new Game({
    levelDef: LEVELS[0],
    input,
    renderer,
    onStateChange: (state, current) => screens.onState(state, current),
});

screens.bind({
    start: () => game.restart(),
    resume: () => game.resume(),
    restart: () => game.restart(),
    pause: () => game.pause(),
    menu: () => game.toMenu(),
});

// Вкладка скрыта — ставим паузу, чтобы игра не шла без игрока.
document.addEventListener('visibilitychange', () => {
    if (document.hidden) game.pause();
});

window.addEventListener('resize', () => renderer.resize());
renderer.resize();
screens.onState('menu', game);
```
